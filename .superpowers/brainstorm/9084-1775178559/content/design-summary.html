3 · Filtering & Sorting Logic 
nowMinutes = currentHour * 60 + currentMinute
dayDelta   = meeting.weekday - nowWeekday

rawDelta = dayDelta * 1440 + meetingMinutes - nowMinutes

// normalize to one week
if rawDelta < 0: rawDelta += 10080
if rawDelta > 10080: rawDelta -= 10080

include if rawDelta <= 180   // next 3 hours
grace:  rawDelta >= -10      // started <10 min ago → still show
  • Sort by rawDelta ascending (soonest first), take first 3
  • 10-minute grace period — meetings started <10 min ago still appear as "In progress"
  • Virtual meetings included in the 3 slots (no distance computed)
  • Known BMLT limitation: no cancellation data — list reflects scheduled meetings only
4 · Data Model 
struct MeetingWithDistance: Identifiable {
    let meeting: Meeting
    let minutesUntil: Int      // negative = in-progress (grace)
    let distanceMiles: Double? // nil for virtual-only

    var id: Int { meeting.id }

    var timeLabel: String {
        if minutesUntil <= 0 { return "In progress" }
        if minutesUntil < 60 { return "in \(minutesUntil) min" }
        let h = minutesUntil / 60, m = minutesUntil % 60
        return m == 0 ? "in \(h) hr" : "in \(h) hr \(m) min"
    }

    var distanceLabel: String? {
        guard let d = distanceMiles else { return nil }
        return String(format: "%.1f mi", d)
    }
}

Wrapper (not extension) keeps Meeting clean as a Codable API type. MeetingWithDistance is ephemeral — recreated on each recompute.
